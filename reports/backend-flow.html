-- 1. Forward trace scan: fetches all spans in a trace in chronological order
CREATE INDEX IF NOT EXISTS idx_events_spans_org_trace_time_asc
  ON events_spans (organization_id, trace_id, occurred_at ASC)
  WHERE trace_id IS NOT NULL;

-- 2. Direct span lookup by ID across the organization
CREATE INDEX IF NOT EXISTS idx_events_spans_org_span_id
  ON events_spans (organization_id, span_id)
  WHERE span_id IS NOT NULL;

-- 3. Database operations explorer: supports filtering by db_system and ordering by duration_ms DESC (slowest first)
CREATE INDEX IF NOT EXISTS idx_events_spans_slow_db
  ON events_spans (organization_id, db_system, duration_ms DESC, occurred_at DESC)
  WHERE is_db_span = TRUE;

-- 4. Database collection queries (e.g. users, tasks, projects)
CREATE INDEX IF NOT EXISTS idx_events_spans_db_collection
  ON events_spans (organization_id, db_collection, occurred_at DESC)
  WHERE db_collection IS NOT NULL;

-- 5. Session correlation: pulls all spans executed under an SDK / user session
CREATE INDEX IF NOT EXISTS idx_events_spans_org_session_time
  ON events_spans (organization_id, session_id, occurred_at DESC)
  WHERE session_id IS NOT NULL;
